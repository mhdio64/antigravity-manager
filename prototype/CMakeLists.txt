cmake_minimum_required(VERSION 3.22)
project(antigravity-release-check VERSION 0.9.2 LANGUAGES CXX)

find_package(Qt6 6.4.2 REQUIRED COMPONENTS Quick Network Concurrent)
find_package(LibArchive REQUIRED)
qt_standard_project_setup()
qt_add_executable(antigravity-release-check main.cpp installer.cpp download.cpp)
qt_add_resources(antigravity-release-check interface PREFIX "/" FILES
    Main.qml icons/antigravity-release-check.png)
target_link_libraries(antigravity-release-check PRIVATE Qt6::Quick Qt6::Network Qt6::Concurrent LibArchive::LibArchive)
target_compile_definitions(antigravity-release-check PRIVATE APP_VERSION="${PROJECT_VERSION}")
install(TARGETS antigravity-release-check RUNTIME DESTINATION bin)
install(FILES antigravity-release-check.desktop DESTINATION share/applications)
install(FILES icons/antigravity-release-check.svg DESTINATION share/icons/hicolor/scalable/apps)
install(FILES icons/antigravity-release-check.png DESTINATION share/icons/hicolor/128x128/apps)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/rtl/node_modules/@electron/asar/package.json")
    message(FATAL_ERROR "Install the pinned RTL dependencies with npm --prefix prototype/rtl ci --omit=dev --ignore-scripts")
endif()
install(DIRECTORY rtl/ DESTINATION share/antigravity-release-check/rtl
    PATTERN ".bin" EXCLUDE)

set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_CONTACT "Antigravity Manager contributors")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "qt6-qpa-plugins, qml6-module-qtqml, qml6-module-qtquick, qml6-module-qtquick-window, qml6-module-qtquick-controls, qml6-module-qtquick-templates, qml6-module-qtquick-dialogs, qml6-module-qtquick-layouts, qml6-module-qtqml-workerscript")
set(PACKAGE_VARIANT "" CACHE STRING "Optional suffix identifying the distribution build")
if(PACKAGE_VARIANT)
    set(CPACK_DEBIAN_FILE_NAME "antigravity-release-check_${PROJECT_VERSION}_${PACKAGE_VARIANT}_amd64.deb")
else()
    set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
endif()
target_compile_definitions(antigravity-release-check PRIVATE PACKAGE_VARIANT="${PACKAGE_VARIANT}")
include(CPack)
